break Bad request from client
Server -> Client: 400\n{"message": "Error: bad request" }
end

// server to handler. send information in form of JSON
Server -> Handler: {"username":" ", "password":" "}

// handler: deserialize into objects. call service method to send objects
Handler -> Service: login(LoginRequest)

// service. call DAO methods for logging in.

// Service calls DAO method for getting user data
Service ->DataAccess: getUser(username)

// DAO manipulates DB
DataAccess->db: Find UserData by username


break Nonexistent User
// break case: return null-no user of that name.
DataAccess-->Service: null
Service-->Server: NonexistentUserException
Server-->Client: 401\n{ "message": "Error: unauthorized"}
end

break Unauthorized Credentials
// break case: return null-no user of that name.
DataAccess-->Service: UserData
Service-->Server: UnauthorizedCredentialsException
Server-->Client: 401\n{ "message": "Error: unauthorized"}
end

// happy path. We get the user data, and it is all correct.
DataAccess-->Service: UserData
// service verifies it, creates an AuthToken, then pushes it via createAuth.
Service->DataAccess: createAuth(authData)
DataAccess->db: Add AuthData

// Now that the new authData has been added, we return a LoginResponse, which contains a username and authToken. This is serialized by the handler and is pushed back to the client via a HTTP response.
Service-->Handler: LoginResponse
Handler-->Server: {"username:" ", "authToken":" "}
Server-->Client: 200\n{"username:" ", "authToken":" "}

end

group#green #lightgreen Logout
Client -> Server: [DELETE] /session\nauthToken

// pass auth token to handler.
Server->Handler: authToken

// handler calls service method Logout.
Handler->Service: logout(LogoutRequest)

// get authData associated with token.
Service->DataAccess: getAuth(authToken)
DataAccess->db: Find AuthData by authToken

// null means that this person was not authorized, and thus cannot log out.
// we do not need to verify password, because the authtoken suggests that the user is logged in. Plus, we cannot deduce the authtoken from the username, anyway. 
break No AuthData associated with authToken
// break case: return null-no user of that name.
DataAccess-->Service: null
Service-->Server: NonExistentAuthException
Server-->Client: 401\n{ "message": "Error: unauthorized"}
end

// we get the auth data
DataAccess-->Service: authData

// service calls DAO to delete auth.
Service->DataAccess: deleteAuth(authData)
DataAccess->db: Remove AuthData

// this may be void, or a boolean.
Service-->Handler: LogoutResponse
// nothing to return, so the handler just returns an empty json object.
Handler-->Server: { }
Server-->Client: 200\n{ }

end
